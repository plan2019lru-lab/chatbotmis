นำเข้าข้อมูลคำถาม (Import JSON)
นำเข้าไฟล์ .json เพื่อกู้คืนคำถาม หรือนำข้อมูลจากเครื่องอื่นมาใส่ในระบบนี้
☁️ การเชื่อมต่อบันทึกคำถามลง Google Sheets อัตโนมัติ (สำหรับ Google Apps Script)
หากคุณเปิดใช้งานแชทบอทนี้ผ่าน Google Apps Script Web App ระบบสามารถส่งประวัติคำถามของผู้ใช้งานทุกข้อไปบันทึกลงใน Google Sheet ของคุณได้แบบ Real-time โดยเพิ่มโค้ดฟังก์ชัน doPost(e) ในไฟล์ Code.gs ดังนี้:
// ฟังก์ชันใน Code.gs สำหรับรับข้อมูลคำถามบันทึกลง Google Sheet function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); sheet.appendRow([new Date(), data.query || '', data.status || '', data.page || '-', data.role || '']); return ContentService.createTextOutput(JSON.stringify({result: 'success'})).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService.createTextOutput(JSON.stringify({result: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON); } }
